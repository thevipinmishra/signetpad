The action is safe to import from a SvelteKit route. It only binds pointer events after the canvas mounts in the browser.

## Install

```bash
pnpm add signetpad
```

## Draw, undo, and clear

<LivePreview client:load example="history" />

```svelte
<script lang="ts">
  import { createSignaturePadAction } from 'signetpad/svelte';

  const { action, controller, snapshot } = createSignaturePadAction({
    viewport: { width: 600, height: 240 },
  });
</script>

<canvas use:action width="600" height="240" aria-label="Agreement signature"></canvas>
<button type="button" disabled={!$snapshot.canUndo} on:click={() => controller.undo()}>
  Undo
</button>
<button type="button" disabled={!$snapshot.canRedo} on:click={() => controller.redo()}>
  Redo
</button>
<button type="button" disabled={$snapshot.isEmpty} on:click={() => controller.clear()}>
  Clear
</button>
```

`$snapshot` follows the Svelte store contract, so you do not subscribe by hand.

## Stroke style

<LivePreview client:load example="stroke" />

```svelte
<script lang="ts">
  import { createSignaturePadAction } from 'signetpad/svelte';

  let color = '#102935';
  const { action, controller } = createSignaturePadAction({
    stroke: { color, width: 3 },
  });

  function onColor(event: Event) {
    const next = (event.currentTarget as HTMLInputElement).value;
    color = next;
    controller.setStrokeStyle({ color: next });
  }
</script>

<canvas use:action aria-label="Styled signature"></canvas>
<input type="color" value={color} on:input={onColor} />
```

If you attach the action to a non-canvas element, capture still works and you paint with your own renderer by subscribing to `controller`.
